% Imatges dins dels callouts, centrades (TODO.md). Una imatge sense peu és un paràgraf que
% LaTeX alinea a l'esquerra; fora dels callouts totes les figures porten peu i ja surten
% centrades. El ganxo marca l'interior dels tcolorbox (els callouts de Quarto), i només s'hi
% centra la imatge que comença un paràgraf (mode vertical): una imatge dins d'una línia de
% text no es toca. Pandoc escriu les imatges de dues maneres, i totes dues hi passen:
% \pandocbounded{\includegraphics…} sense amplada, i \includegraphics[width=…] amb amplada.
\makeatletter
\AddToHook{env/tcolorbox/begin}{\let\ec@dinscallout\relax}
\newcommand\ec@centra[1]{%
  \ifvmode
    \ifdefined\ec@dinscallout
      {\centering#1\par}%
    \else
      #1%
    \fi
  \else
    #1%
  \fi}
\AtBeginDocument{%
  \NewCommandCopy\ec@pandocbounded\pandocbounded
  \RenewDocumentCommand\pandocbounded{m}{\ec@centra{\ec@pandocbounded{#1}}}%
  \NewCommandCopy\ec@includegraphics\includegraphics
  \RenewDocumentCommand\includegraphics{s O{} m}{%
    \ec@centra{\IfBooleanTF{#1}{\ec@includegraphics*[#2]{#3}}{\ec@includegraphics[#2]{#3}}}}%
}
\makeatother
